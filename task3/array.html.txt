    <title>Sum Numbers</title>

<body>

    <button onclick="calculateSum()">Click me</button>

    <p id="result"></p>

    <script>
        function calculateSum() {
            let numbers = [];

            for (let i = 0; i <= 100; i++) {
                numbers.push(i);
            }

            let sum = 0;

            for (let i = 0; i < numbers.length; i++) {
                sum += numbers[i];
            }

            
            document.getElementById("result").innerHTML = "Sum = " + sum;
        }
    </script>
