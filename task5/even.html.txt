    <button onclick="calculateEvenSum()">Click me</button>

    <p id="result"></p>

    <script>
        function calculateEvenSum() {

            let numbers = [];

        
            for (let i = 0; i <= 100; i++) {
                numbers.push(i);
            }

            let sum = 0;

            for (let i = 0; i < numbers.length; i++) {

                if (numbers[i] % 2 == 0) {
                    sum += numbers[i];
                }

            }

            document.getElementById("result").innerHTML =
                "Sum of even numbers = " + sum;
        }
    </script>
